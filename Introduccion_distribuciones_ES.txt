% ---------- Página 4 ----------
\begin{frame}{¿Por qué necesitamos la probabilidad?}
  \vspace{0.6cm}
  \begin{center}
    \colorbox{boxpurple}{\parbox{0.85\textwidth}{\centering\vspace{3pt}
    \textcolor{yellow}{\textbf{la aleatoriedad y la probabilidad son
    fundamentales en la estadística}}\vspace{3pt}}}
  \end{center}
  \vspace{0.6cm}
  \begin{itemize}
    \item los datos son resultados de experimentos y/o sistemas complejos
    \item los experimentos y los sistemas complejos son aleatorios (no
    reproducibles, ``impredecibles'')
    \item necesitamos representar la incertidumbre inherente a la
    naturaleza de los datos
    \item la probabilidad es una medida de la incertidumbre
  \end{itemize}
\end{frame}
